// =============================================
// 1. Core & Listing Metrics
// =============================================

Total Listings = 
DISTINCTCOUNT(airbnb_cleaned[id])

Available Listings % = 
DIVIDE(
    AVERAGE(airbnb_cleaned[availability 365]),
    365
)

Average Availability days = 
AVERAGE(airbnb_cleaned[availability 365])

Average Minimum Nights = 
AVERAGE(airbnb_cleaned[minimum nights])

Average Construction Year = 
AVERAGE(airbnb_cleaned[Construction year])


// =============================================
// 2. Price & Fee Metrics
// =============================================

Average Price = 
AVERAGE(airbnb_cleaned[price])

Average Service Fee = 
AVERAGE(airbnb_cleaned[service fee])


// =============================================
// 3. Host Metrics
// =============================================

Total Hosts = 
DISTINCTCOUNT(airbnb_cleaned[host id2])

Average Listings Per Host = 
DIVIDE( [Total Listings], [Total Hosts])

Verified Hosts = 
CALCULATE( DISTINCTCOUNT(airbnb_cleaned[host id]),  airbnb_cleaned[host_identity_verified] = "verified")

Verified Hosts % = 
DIVIDE( [Verified Hosts], [Total Hosts],  0)



// =============================================
// 4. Reviews & Rating Metrics
// =============================================

Average Number of Reviews = 
AVERAGE(airbnb_cleaned[number of reviews])

Average Rating ⭐ = 
AVERAGE(airbnb_cleaned[review rate number])

Listings With Reviews = 
DIVIDE([Listings With Reviews], [Total Listings],  0)

Listings Without Reviews = 
CALCULATE(COUNTROWS(airbnb_cleaned),airbnb_cleaned[review_status] = "No Reviews")

Listings With Reviews % = 
DIVIDE( [Listings With Reviews],   [Total Listings], 0)


// =============================================
// 5. Booking & House Rules Features
// =============================================

// Instant Bookable
Instant Bookable Listings = 
CALCULATE(
    COUNTROWS(airbnb_cleaned),
    airbnb_cleaned[instant_bookable] = TRUE()
)

Instant Bookable % = 
DIVIDE(
    [Instant Bookable Listings],
    [Total Listings],
    0
)
